create table tb_usuario (
	id_usuario serial primary key,
	tx_nome varchar(80),
	nr_idade numeric(18,0) default 18,
	tx_email varchar(200) not null,
	tx_senha varchar(200) not null,
	dt_cadastro timestamp default current_date
);
create table tb_produto (
	id_produto serial primary key,
	tx_nome varchar(80) not null,
	nr_quantidade numeric(10,0) not null, 
	vl_valor numeric (10,2) not null
);
create table tb_venda (
    cd_venda serial primary key,
    cd_produto integer references tb_produto(id_produto),
    nr_quantidade numeric(10,0)not null,
    vl_valor_unitario numeric(10,2)not null,
    vl_valor_total numeric(10,2)not null,
    dt_venda date default current_date
);

create table tb_aluno(
	id_aluno serial,
	tx_nome varchar(255),
	tx_email varchar(255),
	nr_telefone numeric(18,0),

	constraint pk_tb_aluno primary key(id_aluno)
);
create table tb_professor(
	id_professor serial,
	tx_nome varchar(255),
	tx_email varchar(255),
	nr_telefone numeric(18,0),

	constraint pk_tb_professor primary key(id_professor)
);
create table tb_materia(
	id_materia serial,
	tx_nome varchar(255),
	cd_professor integer,
	nr_cargahoraria numeric(18,0),

	constraint pk_tb_materia primary key(id_materia),
	constraint fk_tb_materia_cd_professor foreign key(cd_professor) references tb_professor(id_professor)
);
create table tb_notas(
	id_notas serial,
	cd_professor integer,
	cd_materia integer,
	vl_prova1 numeric(10,2),
	vl_prova2 numeric(10,2),
	vl_trabalho numeric(10,2),

	constraint pk_tb_notas primary key(id_notas),
	constraint fk_tb_notas_cd_professor foreign key(cd_professor) references tb_professor(id_professor),
	constraint fk_tb_notas_cd_materia foreign key(cd_materia) references tb_materia(id_materia)

);

select
    tc.tx_nome as nome,
    tc.tx_cidade as cidade,
    sum(tiv.nr_quantidade * tiv.vl_unitario) as total_vendas
from tb_cliente tc
left join tb_venda tv
    on tv.cd_cliente = tc.id_cliente
left join tb_item_venda tiv
    on tiv.cd_venda = tv.id_venda
group by
    tc.id_cliente,
    tc.tx_nome,
    tc.tx_cidade
order by total_vendas desc;

select

left join tb_produto tp
    on tp.cd_categoria = tc.id_categoria
where tp.id_produto is null
order by tc.tx_nome asc;

select 
	tc.tx_nome as nomeCliente,
	te.tx_cidade as cidadeCliente,
	te.tx_estado as estadoCliente, 
	email.tx_contato as email,
	telefone.tx_contato as telefone,
	celular.tx_contato as celular
from tb_cliente tc 	
inner join tb_contato email 
	on email.cd_cliente = tc.id_cliente 
	and email.cd_tipo_contato = 1
inner join tb_contato telefone 
	on telefone.cd_cliente = tc.id_cliente 
	and telefone.cd_tipo_contato = 2
	inner join tb_contato celular 
	on celular.cd_cliente = tc.id_cliente 
	and celular.cd_tipo_contato = 3
inner join tb_endereco te 
	on te.id_endereco = tc.cd_endereco 
order by tc.tx_nome asc;

select
    tc.tx_descricao as tipo_contato,
    count(ct.cd_cliente) as quantidade
from tb_tipo_contato tc
left join tb_contato ct
    on ct.cd_tipo_contato = tc.id_tipo_contato
    and ct.fl_ativo = true
group by tc.id_tipo_contato, tc.tx_descricao
order by tc.id_tipo_contato;
	
select
    c.tx_nome as nome,
    max(v.dt_venda) as data_compra_mais_recente
from tb_cliente c
join tb_venda v
    on v.cd_cliente = c.id_cliente
where v.tx_status = 'PAGO'
group by c.id_cliente, c.tx_nome
order by data_compra_mais_recente desc;
