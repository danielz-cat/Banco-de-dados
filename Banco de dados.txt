select
    tb_empresa.tx_razao_social as razao_social,
    resultado.quantidade_vendas_pagas,
    resultado.receita_bruta_total
from tb_empresa
inner join (
 
select
        tb_venda.cd_empresa,
        count(*) as quantidade_vendas_pagas,
        sum(resultado_venda.valor_venda) as receita_bruta_total
    from tb_venda
    inner join (
        select
            tb_item_venda.cd_venda,
            sum(
                tb_item_venda.nr_quantidade *
                tb_item_venda.vl_unitario
            ) as valor_venda
        from tb_item_venda
        group by tb_item_venda.cd_venda
    ) as resultado_venda
        on resultado_venda.cd_venda = tb_venda.id_venda
    where tb_venda.tx_status = 'pago'
    group by tb_venda.cd_empresa
) as resultado
    on resultado.cd_empresa = tb_empresa.id_empresa
where resultado.quantidade_vendas_pagas > 3
  and resultado.receita_bruta_total > 10000
order by resultado.receita_bruta_total desc;

with preco as (
select
sum(item.nr_quantidade * item.vl_unitario ) precoBruto,
ven.cd_empresa
from
tb_venda ven
inner join tb_item_venda item
on item.cd_venda = ven.id_venda
where upper(ven.tx_status) = upper('PAGO')
group by ven.cd_empresa) ,
vendas as (
select
count(id_venda) totalvendas,
ven.cd_empresa
from
tb_venda ven
where upper(ven.tx_status) = upper('PAGO')
group by ven.cd_empresa
)
select * from tb_empresa
inner join vendas on vendas.cd_empresa = id_empresa
inner join preco on preco.cd_empresa = id_empresa
where
precoBruto > 10000
and totalvendas > 3
order by precoBruto desc

with vendas as (
select
	count(distinct venda.id_venda) as nr_compras,
	max(venda.dt_venda ) as dt_ultimacompra,
	min(venda.dt_venda ) as dt_primeiracompra,
	sum(prod.nr_quantidade * prod.vl_unitario) as vl_totalgasto,
	venda.cd_cliente as cliente
from tb_venda venda
inner join tb_item_venda prod
on prod.cd_venda = venda.id_venda
where
	upper(venda.tx_status) = upper('pago')
group by venda.cd_cliente
)
select
	cliente.tx_nome as nome,
	ende.tx_cidade as cidade,
	vendas.dt_primeiracompra,
	vendas.dt_ultimacompra,
	vendas.vl_totalgasto,
	vendas.nr_compras,
case
when vendas.nr_compras < 1 then 'CLIENTE PEQUENO'
when vendas.nr_compras = 1 then 'CLIENTE MEDIO'
when vendas.nr_compras = 2 then 'CLIENTE GRANDE'
else 'OUTROS'
end as status,
(
select
	p.tx_nome as produtoFavorito
from
	tb_venda venda
inner join tb_item_venda prod on
	prod.cd_venda = venda.id_venda
inner join tb_produto p on p.id_produto = prod.cd_produto
where
	upper(venda.tx_status) = upper('pago')
and cliente.id_cliente = venda.cd_cliente
group by p.tx_nome, p.id_produto
order by sum(prod.nr_quantidade) desc
limit 1
) as produtoFavorito2
from tb_cliente cliente
inner join tb_endereco ende on ende.id_endereco = cliente.cd_endereco
inner join vendas on vendas.cliente = cliente.id_cliente
